cmake_minimum_required(VERSION 3.16)
project(Mochi VERSION 0.3.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

add_library(Mochi STATIC src/Animator.cpp src/Character.cpp src/Input.cpp src/Mascot.cpp src/Motion.cpp src/Renderer.cpp src/Surface.cpp)
add_library(Mochi::Mochi ALIAS Mochi)
target_include_directories(Mochi PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>"
  "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
target_compile_features(Mochi PUBLIC cxx_std_17)

install(TARGETS Mochi EXPORT MochiTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY src/mochi DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.h")
install(FILES src/Mochi.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/Mochi)
set(MOCHI_CONFIG_DIR "${CMAKE_INSTALL_LIBDIR}/cmake/Mochi")
configure_package_config_file(cmake/MochiConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/MochiConfig.cmake" INSTALL_DESTINATION ${MOCHI_CONFIG_DIR})
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MochiConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT MochiTargets NAMESPACE Mochi:: DESTINATION ${MOCHI_CONFIG_DIR})
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MochiConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/MochiConfigVersion.cmake" DESTINATION ${MOCHI_CONFIG_DIR})

option(MOCHI_BUILD_TESTS "Build the portable SDK tests" OFF)
option(MOCHI_BUILD_EXAMPLE "Build the portable examples" OFF)
if(MOCHI_BUILD_TESTS)
  enable_testing()
  add_executable(mochi_tests tests/sdk_test.cpp)
  target_link_libraries(mochi_tests PRIVATE Mochi::Mochi)
  # Tests use assert, including when this library is built in Release mode.
  if(MSVC)
    target_compile_options(mochi_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(mochi_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME mochi_sdk COMMAND mochi_tests)
  add_executable(mochi_integration_tests tests/integration_test.cpp)
  target_link_libraries(mochi_integration_tests PRIVATE Mochi::Mochi)
  find_package(Threads REQUIRED)
  add_executable(mochi_queue_tests tests/queue_test.cpp)
  target_link_libraries(mochi_queue_tests PRIVATE Mochi::Mochi Threads::Threads)
  foreach(test_target mochi_integration_tests mochi_queue_tests)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
  add_test(NAME mochi_integration COMMAND mochi_integration_tests)
  add_test(NAME mochi_queue COMMAND mochi_queue_tests)
endif()
if(MOCHI_BUILD_EXAMPLE)
  add_executable(mochi_offscreen examples/offscreen.cpp)
  target_link_libraries(mochi_offscreen PRIVATE Mochi::Mochi)
  add_executable(mochi_cast examples/cast.cpp)
  target_link_libraries(mochi_cast PRIVATE Mochi::Mochi)
  add_executable(mochi_application examples/application.cpp)
  target_link_libraries(mochi_application PRIVATE Mochi::Mochi)
  add_executable(mochi_stripes examples/stripes.cpp)
  target_link_libraries(mochi_stripes PRIVATE Mochi::Mochi)
endif()
