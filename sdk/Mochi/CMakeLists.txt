cmake_minimum_required(VERSION 3.16)
project(Mochi VERSION 0.2.0 LANGUAGES CXX)

add_library(Mochi src/Animator.cpp src/Character.cpp src/Input.cpp src/Motion.cpp src/Renderer.cpp)
add_library(Mochi::Mochi ALIAS Mochi)
target_include_directories(Mochi PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(Mochi PUBLIC cxx_std_17)

option(MOCHI_BUILD_TESTS "Build the portable SDK tests" OFF)
option(MOCHI_BUILD_EXAMPLE "Build the offscreen example" OFF)
if(MOCHI_BUILD_TESTS)
  enable_testing()
  add_executable(mochi_tests tests/sdk_test.cpp)
  target_link_libraries(mochi_tests PRIVATE Mochi::Mochi)
  # Tests use assert, including when this library is built in Release mode.
  if(MSVC)
    target_compile_options(mochi_tests PRIVATE /UNDEBUG)
  else()
    target_compile_options(mochi_tests PRIVATE -UNDEBUG)
  endif()
  add_test(NAME mochi_sdk COMMAND mochi_tests)
endif()
if(MOCHI_BUILD_EXAMPLE)
  add_executable(mochi_offscreen examples/offscreen.cpp)
  target_link_libraries(mochi_offscreen PRIVATE Mochi::Mochi)
  add_executable(mochi_cast examples/cast.cpp)
  target_link_libraries(mochi_cast PRIVATE Mochi::Mochi)
endif()
